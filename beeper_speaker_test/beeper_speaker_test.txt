// Ultra-simple PAM8403 + speaker test using GPIO25 square wave.
// This bypasses all MP3/HTTP logic and just drives the amp input directly.

const int audioPin = 25; // wired to PAM8403 input via resistor/cap

void setup() { pinMode(audioPin, OUTPUT); }

void loop() {
  // Beep at ~1 kHz for 1 second
  unsigned long start = millis();
  while (millis() - start < 1000) {
    digitalWrite(audioPin, HIGH);
    delayMicroseconds(500);
    digitalWrite(audioPin, LOW);
    delayMicroseconds(500);
  }

  // Then stay silent for 2 seconds
  digitalWrite(audioPin, LOW);
  delay(2000);
}
